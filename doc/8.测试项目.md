### 测试金字塔

```
        ┌─────────┐
        │  E2E    │  少量，核心业务流程
        ├─────────┤
        │ 集成测试  │  模块间交互
        ├─────────┤
        │ 接口测试  │  API 契约验证
        ├─────────┤
```

### 测试覆盖目标

| 层级 | 覆盖率目标 | 工具 |
|------|-----------|------|
| 后端单元测试 | Service 层 ≥ 80% | JUnit 5 + Mockito |
| 接口测试 | 核心接口 100% | Postman / REST Assured |
| 前端组件测试 | 公共组件 ≥ 60% | Vitest + Vue Test Utils |
| E2E 测试 | 核心流程全覆盖 | Playwright / Cypress |

### 测试分类

| 类型 | 说明 | 执行时机 |
|------|------|----------|
| 冒烟测试 | 验证核心功能可用 | 每次部署后 |
| 回归测试 | 验证修改未引入新问题 | 每次发版前 |
| 安全测试 | SQL 注入、XSS、权限绕过 | 发版前 + 季度 |
| 性能测试 | 接口响应时间、并发能力 | 重大版本前 |

## 单元测试

### 后端 Service 测试

```java
// src/test/java/com/company/rbac/service/SysUserServiceTest.java
@SpringBootTest
@ActiveProfiles("test")
class SysUserServiceTest {

    @Autowired
    private ISysUserService userService;

    @MockBean
    private SysUserMapper userMapper;

    @Test
    @DisplayName("分页查询用户列表 - 正常返回")
    void listUsers_shouldReturnPageResult() {
        // Given
        UserQueryDTO query = new UserQueryDTO();
        query.setPageNum(1);
        query.setPageSize(10);

        List<SysUser> mockUsers = List.of(buildUser(1L, "admin"));
        when(userMapper.selectUserList(any())).thenReturn(mockUsers);
        when(userMapper.selectUserCount(any())).thenReturn(1L);

        // When
        PageResult<SysUserVO> result = userService.listUsers(query);

        // Then
        assertThat(result.getTotal()).isEqualTo(1);
        assertThat(result.getRecords()).hasSize(1);
        assertThat(result.getRecords().get(0).getUserName()).isEqualTo("admin");
    }

    @Test
    @DisplayName("创建用户 - 用户名重复抛异常")
    void createUser_duplicateName_shouldThrowException() {
        // Given
        SysUserDTO dto = new SysUserDTO();
        dto.setUserName("admin");

        when(userMapper.selectByUserName("admin")).thenReturn(new SysUser());

        // When & Then
        assertThatThrownBy(() -> userService.createUser(dto))
            .isInstanceOf(BusinessException.class)
            .hasMessageContaining("用户名已存在");
    }

    @Test
    @DisplayName("删除用户 - 不能删除超级管理员")
    void deleteUsers_adminUser_shouldThrowException() {
        assertThatThrownBy(() -> userService.deleteUsers(new Long[]{1L}))
            .isInstanceOf(BusinessException.class)
            .hasMessageContaining("不能删除超级管理员");
    }

    private SysUser buildUser(Long id, String userName) {
        SysUser user = new SysUser();
        user.setId(id);
        user.setUserName(userName);
        user.setStatus("0");
        return user;
    }
}
```

### 后端 Controller 测试

```java
@SpringBootTest
@AutoConfigureMockMvc
@ActiveProfiles("test")
class SysUserControllerTest {

    @Autowired
    private MockMvc mockMvc;

    @MockBean
    private ISysUserService userService;

    private String token;

    @BeforeEach
    void setUp() {
        token = "Bearer test-token";
    }

    @Test
    @DisplayName("GET /system/user/list - 返回用户列表")
    @WithMockUser(authorities = "system:user:list")
    void listUsers_shouldReturn200() throws Exception {
        PageResult<SysUserVO> pageResult = new PageResult<>();
        pageResult.setRecords(List.of());
        pageResult.setTotal(0);

        when(userService.listUsers(any())).thenReturn(pageResult);

        mockMvc.perform(get("/system/user/list")
                .param("pageNum", "1")
                .param("pageSize", "10")
                .header("Authorization", token))
            .andExpect(status().isOk())
            .andExpect(jsonPath("$.code").value(200))
            .andExpect(jsonPath("$.data.records").isArray());
    }

    @Test
    @DisplayName("GET /system/user/list - 无权限返回 403")
    void listUsers_noPermission_shouldReturn403() throws Exception {
        mockMvc.perform(get("/system/user/list")
                .header("Authorization", token))
            .andExpect(status().isForbidden());
    }
}
```

```java
// src/test/java/com/company/rbac/api/UserApiTest.java
@SpringBootTest(webEnvironment = SpringBootTest.WebEnvironment.RANDOM_PORT)
@ActiveProfiles("test")
class UserApiTest {

    @LocalServerPort
    private int port;

    private String baseUrl;
    private String token;

    @BeforeEach
    void setUp() {
        baseUrl = "http://localhost:" + port + "/api";
        token = loginAndGetToken("admin", "admin123");
    }

    @Test
    void listUsers_withValidToken_shouldReturn200() {
        given()
            .header("Authorization", "Bearer " + token)
            .queryParam("pageNum", 1)
            .queryParam("pageSize", 10)
        .when()
            .get(baseUrl + "/system/user/list")
        .then()
            .statusCode(200)
            .body("code", equalTo(200))
            .body("data.records", notNullValue());
    }

    @Test
    void createUser_withDuplicateName_shouldReturn1001() {
        String body = """
            {"userName": "admin", "nickName": "重复", "password": "123456"}
            """;

        given()
            .header("Authorization", "Bearer " + token)
            .contentType(ContentType.JSON)
            .body(body)
        .when()
            .post(baseUrl + "/system/user")
        .then()
            .statusCode(200)
            .body("code", equalTo(1001));
    }
}
```

### 权限指令测试

```typescript
// src/directives/__tests__/permission.spec.ts
import { describe, it, expect, vi } from 'vitest'
import { hasPermi } from '../permission'

vi.mock('@/store/modules/user', () => ({
  useUserStore: () => ({
    permissions: ['system:user:list', 'system:user:add']
  })
}))

describe('v-hasPermi directive', () => {
  it('should keep element when permission exists', () => {
    const el = document.createElement('button')
    const parent = document.createElement('div')
    parent.appendChild(el)

    hasPermi.mounted(el, { value: ['system:user:add'] })
    expect(parent.contains(el)).toBe(true)
  })

  it('should remove element when permission missing', () => {
    const el = document.createElement('button')
    const parent = document.createElement('div')
    parent.appendChild(el)

    hasPermi.mounted(el, { value: ['system:user:remove'] })
    expect(parent.contains(el)).toBe(false)
  })
})
```

## 集成测试

### 核心业务流程

#### 流程 1：用户登录 → 获取菜单 → 访问页面

```
1. POST /api/auth/login (admin/admin123)         → 获取 Token
2. GET  /api/auth/info                           → 获取用户信息和权限
3. GET  /api/system/menu/routers                → 获取动态路由
4. GET  /api/system/user/list                   → 访问用户管理（有权限）
5. GET  /api/system/role/list                   → 访问角色管理（有权限）
```

#### 流程 2：RBAC 权限验证

```
1. 以普通用户 (user/user123) 登录
2. GET /api/system/user/list    → 预期 403（无权限）
3. GET /api/system/menu/list      → 预期 200（有权限）
4. 管理员为用户分配 system:user:list 权限
5. 普通用户重新登录
6. GET /api/system/user/list    → 预期 200
```

#### 流程 3：代码生成

```
1. 创建 biz_order 表
2. POST /api/tool/gen/importTable  → 导入表
3. PUT  /api/tool/gen              → 配置生成参数
4. GET  /api/tool/gen/preview/{id} → 预览代码
5. GET  /api/tool/gen/download/{id}→ 下载代码包
6. 验证生成的代码可编译通过
```

### GitHub Actions 配置

```yaml
# .github/workflows/test.yml
name: Test

on:
  push:
    branches: [develop, main]
  pull_request:
    branches: [develop]

jobs:
  backend-test:
    runs-on: ubuntu-latest
    services:
      mysql:
        image: mysql:8.0
        env:
          MYSQL_ROOT_PASSWORD: root
          MYSQL_DATABASE: rbac_test
        ports:
          - 3306:3306
      redis:
        image: redis:7-alpine
        ports:
          - 6379:6379

    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with:
          java-version: '17'
          distribution: 'temurin'
      - name: Run Backend Tests
        run: cd backend && mvn test -Dspring.profiles.active=test

  frontend-test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: '20'
      - name: Install & Test
        run: |
          cd frontend
          npm ci
          npm run type-check
          npm run lint
          npm run test
```
